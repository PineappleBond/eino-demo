> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the complete Next.js frontend for Phase 1 — template browser, project management, chat interface with streaming, tool call cards, WebSocket reconnection, IndexedDB persistence, and i18n.

**Architecture:** `applyUpdates()` single entry point for all incoming updates. Topic-based subscription from route context. IndexedDB for entity persistence via HTTP sync. React Context + useReducer for state. Ant Design for all UI components.

**Tech Stack:** Next.js App Router, React 19, Ant Design, next-intl, openapi-typescript, @chenglou/pretext, react-markdown, remark-gfm, react-syntax-highlighter

const API_BASE = process.env.NEXT_PUBLIC_API_BASE || 'http://localhost:8080';

function getToken(): string | null {
  if (typeof window === 'undefined') return null;
  return localStorage.getItem('auth_token');
}

function headers(extra?: Record<string, string>): Record<string, string> {
  const h: Record<string, string> = {
    'Content-Type': 'application/json',
    ...extra,
  };
  const token = getToken();
  if (token) {
    h['Authorization'] = `Bearer ${token}`;
  }
  return h;
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${API_BASE}${path}`, {
    ...init,
    headers: headers(init?.headers as Record<string, string>),
  });

  if (!res.ok) {
    const body = await res.json().catch(() => null);
    const message = body?.error?.message || `HTTP ${res.status}`;
    throw new Error(message);
  }

  if (res.status === 204) return undefined as T;
  return res.json() as Promise<T>;
}

const DB_NAME = 'eino-demo';
const DB_VERSION = 1;
const SETTINGS_STORE = 'settings';

interface SettingsRow {
  key: string;
  value: unknown;
}

let dbPromise: Promise<IDBDatabase> | null = null;

function openDB(): Promise<IDBDatabase> {
  if (dbPromise) return dbPromise;

  dbPromise = new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, DB_VERSION);
    req.onupgradeneeded = () => {
      const db = req.result;
      if (!db.objectStoreNames.contains(SETTINGS_STORE)) {
        db.createObjectStore(SETTINGS_STORE, { keyPath: 'key' });
      }
      // Entity stores will be added in later phases.
      // For Phase 1, we only need settings.latest_seq.
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });

  return dbPromise;
}

async function getSetting<T>(key: string): Promise<T | null> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(SETTINGS_STORE, 'readonly');
    const store = tx.objectStore(SETTINGS_STORE);
    const req = store.get(key);
    req.onsuccess = () => resolve(req.result ? (req.result as SettingsRow).value as T : null);
    req.onerror = () => reject(req.error);
  });
}

async function setSetting(key: string, value: unknown): Promise<void> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(SETTINGS_STORE, 'readwrite');
    const store = tx.objectStore(SETTINGS_STORE);
    store.put({ key, value });
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}

/**
 * Get the latest_seq cursor for seq continuity checking.
 */
export async function getLatestSeq(): Promise<number> {
  const val = await getSetting<number>('latest_seq');
  return val ?? 0;
}

/**
 * Set the latest_seq cursor after processing persistable updates.
 */
export async function setLatestSeq(seq: number): Promise<void> {
  await setSetting('latest_seq', seq);
}

/**
 * Derive topic from route context.
 * Backend has no topic concept — frontend extracts from Update payload.
 */
export function deriveTopic(payload: Record<string, unknown>): string {
  if (payload.conversation_id) {
    return `conv:${payload.conversation_id}`;
  }
  if (payload.project_id) {
    return `project:${payload.project_id}`;
  }
  return 'system';
}

/**
 * Get topic from current route.
 * Called by useSubscribe in page components.
 */
export function topicFromPath(pathname: string): string {
  // /en/chat/[convId] → conv:{convId}
  const match = pathname.match(/\/[a-z]{2}\/chat\/([a-f0-9-]+)/);
  if (match) {
    return `conv:${match[1]}`;
  }
  // /en/chat/ → project or system (determined by project context)
  if (pathname.includes('/chat')) {
    return 'system';
  }
  return 'system';
}

- [ ] **Step 4: Create web/src/lib/updateDispatcher.ts**

```ts
import { getLatestSeq, setLatestSeq } from '../store/indexedDB';
import { deriveTopic } from '../store/topic';

type UpdateType =
  | 'message.new' | 'message.delta' | 'message.done'
  | 'message.tool_call' | 'message.thinking' | 'message.error' | 'message.stop'
  | 'conversation.created' | 'conversation.deleted'
  | 'conversation.compacting' | 'conversation.compacted' | 'conversation.archived'
  | 'project.created' | 'project.deleted'
  | 'settings.changed'
  | 'empty';

interface Update {
  seq: number;
  type: UpdateType;
  payload: Record<string, unknown>;
}

type Subscriber = (update: Update) => void;

class UpdateDispatcher {
  private subscribers = new Map<string, Set<Subscriber>>();
  private processing = false;
  private queue: Update[][] = [];

  /**
   * Subscribe to a topic. Returns unsubscribe function.
   */
  subscribe(topic: string, fn: Subscriber): () => void {
    if (!this.subscribers.has(topic)) {
      this.subscribers.set(topic, new Set());
    }
    this.subscribers.get(topic)!.add(fn);
    return () => {
      this.subscribers.get(topic)?.delete(fn);
    };
  }

  /**
   * Main entry point. ALL incoming updates (WS push or HTTP response) go through this.
   * Thread-safe: concurrent calls are serialized via queue.
   */
  async applyUpdates(updates: Update[]): Promise<void> {
    if (this.processing) {
      this.queue.push(updates);
      return;
    }
    this.processing = true;

    try {
      await this._processBatch(updates);

      while (this.queue.length > 0) {
        const next = this.queue.shift()!;
        await this._processBatch(next);
      }
    } finally {
      this.processing = false;
    }
  }

  private async _processBatch(updates: Update[]): Promise<void> {
    const persistable = updates.filter((u) => u.seq > 0);
    const ephemeral = updates.filter((u) => u.seq === 0);

    if (persistable.length > 0) {
      const localMaxSeq = await getLatestSeq();
      const minPersistable = Math.min(...persistable.map((u) => u.seq));

      if (minPersistable !== localMaxSeq + 1) {
        // Seq gap detected — abort, trigger HTTP pull
        console.warn(`[UpdateDispatcher] seq gap: local=${localMaxSeq}, incoming min=${minPersistable}`);
        return;
      }

      const maxSeq = Math.max(...persistable.map((u) => u.seq));
      await setLatestSeq(maxSeq);
    }

    // Notify subscribers by topic
    for (const update of updates) {
      const topic = deriveTopic(update.payload);
      const subs = this.subscribers.get(topic);
      if (subs) {
        for (const fn of subs) {
          fn(update);
        }
      }
      // Also notify system topic for settings changes
      if (update.type === 'settings.changed') {
        const systemSubs = this.subscribers.get('system');
        if (systemSubs) {
          for (const fn of systemSubs) {
            fn(update);
          }
        }
      }
    }
  }
}

export const dispatcher = new UpdateDispatcher();
export type { Update, UpdateType };
```

'use client';

import { createContext, useContext, useEffect, useRef, useState, ReactNode } from 'react';
import { dispatcher } from '../lib/updateDispatcher';
import { getLatestSeq } from '../store/indexedDB';

interface WSContextValue {
  connected: boolean;
  reconnecting: boolean;
}

const WSContext = createContext<WSContextValue>({ connected: false, reconnecting: false });

export function useWS() {
  return useContext(WSContext);
}

const WS_BASE = process.env.NEXT_PUBLIC_WS_BASE || 'ws://localhost:8080';

function getToken(): string {
  return localStorage.getItem('auth_token') || '';
}

export function WSProvider({ children }: { children: ReactNode }) {
  const [connected, setConnected] = useState(false);
  const [reconnecting, setReconnecting] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectTimerRef = useRef<number | null>(null);
  const reconnectDelayRef = useRef(1000);

  const connect = async () => {
    const token = getToken();
    if (!token) return;

    const lastSeq = await getLatestSeq();
    const url = `${WS_BASE}/ws?token=${encodeURIComponent(token)}&last_seq=${lastSeq}`;
    const ws = new WebSocket(url);
    wsRef.current = ws;

    ws.onopen = () => {
      setConnected(true);
      setReconnecting(false);
      reconnectDelayRef.current = 1000;

      // Start heartbeat
      const heartbeat = setInterval(() => {
        if (ws.readyState === WebSocket.OPEN) {
          ws.send(JSON.stringify({ type: 'ping', payload: {} }));
        } else {
          clearInterval(heartbeat);
        }
      }, 30_000);
    };

    ws.onmessage = (event) => {
      try {
        const frame = JSON.parse(event.data);
        if (frame.type === 'connected') {
          // Server sent max_seq — compare with local
          if (frame.payload.max_seq > 0) {
            // Gap may exist — handled by seq check in applyUpdates
          }
        } else if (frame.type === 'updates') {
          dispatcher.applyUpdates(frame.payload);
        }
      } catch {
        // ignore parse errors
      }
    };

    ws.onclose = () => {
      setConnected(false);
      setReconnecting(true);

      // Exponential backoff reconnect
      const delay = reconnectDelayRef.current;
      reconnectTimerRef.current = window.setTimeout(() => {
        reconnectDelayRef.current = Math.min(delay * 2, 30_000);
        connect();
      }, delay);
    };

    ws.onerror = () => {
      ws.close();
    };
  };

  useEffect(() => {
    connect();
    return () => {
      if (reconnectTimerRef.current) clearTimeout(reconnectTimerRef.current);
      wsRef.current?.close();
    };
  }, []);

  return (
    <WSContext.Provider value={{ connected, reconnecting }}>
      {children}
    </WSContext.Provider>
  );
}

interface UpdateContextValue {
  subscribe: (topic: string, fn: SubscriberFn) => () => void;
  flush: (topic: string) => void;
}

const UpdateContext = createContext<UpdateContextValue>({
  subscribe: () => () => {},
  flush: () => {},
});

export function useUpdateDispatcher() {
  return useContext(UpdateContext);

/**
 * Hook to subscribe to a topic.
 * Mounts subscribe, unmounts unsubscribe.
 */
export function useSubscribe(topic: string, onMessage: (update: Update) => void) {
  const { subscribe } = useUpdateDispatcher();
  useEffect(() => {
    return subscribe(topic, onMessage);

    // Future: replay accumulated updates for this topic
  }, []);

    <UpdateContext.Provider value={{ subscribe, flush }}>
      {children}
    </UpdateContext.Provider>
  );

import { useState, useCallback } from 'react';

export function useStream() {
  const [streamingContent, setStreamingContent] = useState('');
  const [isStreaming, setIsStreaming] = useState(false);

  const appendDelta = useCallback((delta: string) => {
    setStreamingContent((prev) => prev + delta);
  }, []);

  const startStream = useCallback(() => {
    setStreamingContent('');
    setIsStreaming(true);
  }, []);

  const endStream = useCallback((finalContent?: string) => {
    setIsStreaming(false);
    if (finalContent !== undefined) {
      setStreamingContent(finalContent);
    }
  }, []);

  const stopStream = useCallback(() => {
    setIsStreaming(false);
  }, []);

  return {
    streamingContent,
    isStreaming,
    appendDelta,
    startStream,
    endStream,
    stopStream,
  };
}

```tsx
'use client';

import { useEffect, useState, useCallback } from 'react';
import { useParams } from 'next/navigation';
import { Typography } from 'antd';
import { MessageList } from '@/components/chat/MessageList';
import { ChatInput } from '@/components/chat/ChatInput';
import { useStream } from '@/hooks/useStream';
import { useSubscribe } from '@/providers/UpdateProvider';
import { api, Message as MessageT } from '@/lib/api';

export default function ChatPage() {
  const params = useParams() as { locale: string; convId: string };
  const convId = params.convId;
  const [messages, setMessages] = useState<MessageT[]>([]);
  const [loading, setLoading] = useState(true);
  const { streamingContent, isStreaming, appendDelta, startStream, endStream } = useStream();

  // Load messages on mount
  useEffect(() => {
    api.get<{ messages: MessageT[]; has_more: boolean; latest_seq: number }>(
      `/api/v1/conversations/${convId}/messages`
    )
      .then((res) => {
        setMessages(res.messages);
      })
      .catch(console.error)
      .finally(() => setLoading(false));
  }, [convId]);

  // Subscribe to updates for this conversation
  const handleMessage = useCallback((update: any) => {
    switch (update.type) {
      case 'message.delta':
        startStream();
        appendDelta(update.payload.delta || '');
        break;
      case 'message.done':
        endStream();
        setMessages((prev) => [...prev, update.payload.message]);
        break;
      case 'message.new':
        setMessages((prev) => [...prev, update.payload.message]);
        break;
      case 'message.stop':
        endStream(update.payload.content || '');
        break;
      case 'message.error':
        endStream();
        break;
    }
  }, [appendDelta, startStream, endStream]);

  useSubscribe(`conv:${convId}`, handleMessage);

  const handleSend = useCallback(
    (content: string) => {
      startStream();
      api
        .post<any>(`/api/v1/conversations/${convId}/messages`, { content })
        .then((updates) => {
          // HTTP response contains Update[] — pass to dispatcher
          // For now, directly add the user message
          for (const u of updates) {
            if (u.type === 'message.new') {
              setMessages((prev) => [...prev, u.payload.message]);
            }
          }
        })
        .catch((err) => {
          console.error('Send failed:', err);
          endStream();
        });
    },
    [convId, startStream, endStream]
  );

  const handleStop = useCallback(() => {
    api.post(`/api/v1/conversations/${convId}/stop`).catch(console.error);
    endStream();
  }, [convId, endStream]);

  if (loading) return <div style={{ padding: 24 }}>Loading...</div>;

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100vh' }}>
      <Typography.Title level={5} style={{ padding: '12px 24px', borderBottom: '1px solid #f0f0f0', margin: 0 }}>
        Conversation {convId}
      </Typography.Title>
      <MessageList messages={messages} streamingContent={streamingContent} />
      <ChatInput onSend={handleSend} onStop={handleStop} isStreaming={isStreaming} />
    </div>
  );
}

import { useTranslations } from 'next-intl';
import { Card, Form, Select, Radio, message } from 'antd';
import { api, Settings } from '@/lib/api';

export default function SettingsPage() {
  const t = useTranslations('settings');
  const params = useParams() as { locale: string };
  const [form] = Form.useForm();
  const [loading, setLoading] = useState(true);
  const [settings, setSettings] = useState<Settings | null>(null);

  useEffect(() => {
    api.get<Settings>('/api/v1/settings')
      .then((s) => {
        setSettings(s);
        form.setFieldsValue(s);
      })
      .catch(console.error)
      .finally(() => setLoading(false));
  }, [form]);

  const handleFinish = async (values: Partial<Settings>) => {
    try {
      const updated = await api.patch<Settings>('/api/v1/settings', values);
      setSettings(updated);
      message.success(t('saved'));
    } catch (err) {
      message.error('Failed to save settings');
    }
  };

  if (loading || !settings) return <div style={{ padding: 24 }}>Loading...</div>;

  return (
    <div style={{ maxWidth: 600, margin: '0 auto', padding: 24 }}>
      <h2>{t('title')}</h2>
      <Card>
        <Form form={form} layout="vertical" onValueset={() => {
          const values = form.getFieldsValue();
          handleFinish(values);
        }}>
          <Form.Item name="model_tier" label={t('model')}>
            <Select>
              <Select.Option value="haiku">
                haiku — {t('model_desc.haiku')}
              </Select.Option>
              <Select.Option value="sonnet">
                sonnet — {t('model_desc.sonnet')}
              </Select.Option>
              <Select.Option value="opus">
                opus — {t('model_desc.opus')}
              </Select.Option>
            </Select>
          </Form.Item>
          <Form.Item name="locale" label={t('language')}>
            <Radio.Group>
              <Radio.Button value="en">English</Radio.Button>
              <Radio.Button value="zh">中文</Radio.Button>
            </Radio.Group>
          </Form.Item>
          <Form.Item name="theme" label={t('theme')}>
            <Radio.Group>
              <Radio.Button value="light">Light</Radio.Button>
              <Radio.Button value="dark">Dark</Radio.Button>
            </Radio.Group>
          </Form.Item>
        </Form>
      </Card>
    </div>
  );
}
```

- [ ] **Step 2: Create auth entry point**

Create `web/src/app/[locale]/login/page.tsx`:

```tsx
'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { useTranslations } from 'next-intl';
import { Card, Input, Button, message } from 'antd';

export default function LoginPage() {
  const t = useTranslations('auth');
  const router = useRouter();
  const [token, setToken] = useState('');

  const handleConnect = () => {
    if (!token.trim()) {
      message.error('Token is required');
      return;
    }
    localStorage.setItem('auth_token', token.trim());
    router.push('/en/');
  };

  return (
    <div style={{ display: 'flex', justifyContent: 'center', alignItems: 'center', minHeight: '100vh', background: '#f5f5f5' }}>
      <Card title={t('tokenLabel')} style={{ width: 400 }}>
        <Input.Password
          value={token}
          onChange={(e) => setToken(e.target.value)}
          placeholder={t('tokenPlaceholder')}
          onPressEnter={handleConnect}
          style={{ marginBottom: 16 }}
        />
        <Button type="primary" block onClick={handleConnect}>
          {t('connect')}
        </Button>
      </Card>
    </div>
  );
}
```
